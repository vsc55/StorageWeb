{{ _("Example for nginx with these values") }} 
server {
    listen 443 ssl;
    http2 on;
    server_name {{ example.host }};
    server_tokens off;
    ssl_certificate     /etc/letsencrypt/live/{{ example.host }}/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/{{ example.host }}/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;

    client_max_body_size {{ example.body_mb }}m;
    proxy_request_buffering off;
    proxy_buffering off;
    proxy_max_temp_file_size 0;
    proxy_read_timeout 300s;
    proxy_send_timeout 300s;

    location {{ example.prefix or '' }}/ {
        proxy_pass http://{{ example.upstream }}/;
        proxy_set_header Host              $host;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host  $host;
        proxy_set_header X-Forwarded-Prefix "{{ example.prefix or '' }}";
    }
}

server {
    listen 80;
    server_name {{ example.host }};
    return 301 https://$host$request_uri;
}
